아키텍처 아이디어
1. Spring Boot Starter 형태로 제공
temporal-spring-boot-starter라는 이름의 라이브러리를 만든다고 가정해 봅시다. 이 Starter를 추가하면 Spring Boot 애플리케이션 구동 시 다음과 같은 일이 자동으로 일어납니다.

application.yml 또는 application.properties에 정의한 Temporal 설정(서비스 주소, namespace, task queue, 재시도 정책 등)을 읽어 WorkflowServiceStubs, WorkflowClient, WorkerFactory를 자동 설정.
특정 패키지 내의 워크플로우 인터페이스와 구현체, 액티비티 구현체를 스캔하고 자동으로 워커 등록.
@EnableTemporalWorkflows 어노테이션을 사용하면, 해당 패키지 경로에서 워크플로우와 액티비티 인터페이스를 찾고 자동으로 빈을 등록하고 워커를 시작.
2. 어노테이션 및 인터페이스 추상화
@TemporalWorkflow 어노테이션: 워크플로우 인터페이스에 붙여서 해당 인터페이스가 워크플로우임을 선언.
@TemporalActivity 어노테이션: 액티비티 인터페이스나 구현 클래스에 붙여서 해당 액티비티를 자동 등록.
@WorkflowMethod, @ActivityMethod 는 Temporal SDK 어노테이션을 그대로 사용하거나, 별도로 정의하여 더욱 단순화.

코드 복사
@Configuration
@ConditionalOnProperty(prefix = "temporal", name = "service.address")
public class TemporalAutoConfiguration {

    @Autowired
    private Environment environment;

    @Bean
    @ConditionalOnMissingBean
    public TemporalProperties temporalProperties() {
        // Environment에서 temporal.* 설정 로딩
        return TemporalProperties.fromEnvironment(environment);
    }

    @Bean
    @ConditionalOnMissingBean
    public WorkflowServiceStubs workflowServiceStubs(TemporalProperties props) {
        return WorkflowServiceStubs.newInstance(
            WorkflowServiceStubsOptions.newBuilder()
                .setTarget(props.getServiceAddress())
                .build()
        );
    }

    @Bean
    @ConditionalOnMissingBean
    public WorkflowClient workflowClient(TemporalProperties props, WorkflowServiceStubs stubs) {
        return WorkflowClient.newInstance(
            stubs,
            WorkflowClientOptions.newBuilder()
                .setNamespace(props.getNamespace())
                .build()
        );
    }

    @Bean
    @ConditionalOnMissingBean
    public WorkerFactory workerFactory(WorkflowClient client) {
        return WorkerFactory.newInstance(client);
    }

    @Bean
    @ConditionalOnMissingBean
    public TemporalWorkerRegistrar temporalWorkerRegistrar(WorkerFactory factory, TemporalProperties props) {
        return new TemporalWorkerRegistrar(factory, props.getTaskQueue());
    }

    // 워크플로우 인터페이스 스캔, 액티비티 구현체 스캔을 하는 PostProcessor 등록
    @Bean
    public static TemporalBeanPostProcessor temporalBeanPostProcessor(
        ApplicationContext context, TemporalWorkerRegistrar registrar) {
        return new TemporalBeanPostProcessor(context, registrar);
    }

    @Bean
    public TemporalWorkerStarter temporalWorkerStarter(WorkerFactory factory) {
        return new TemporalWorkerStarter(factory);
    }
}
5. 빈 후처리기 (TemporalBeanPostProcessor) 예시
BeanPostProcessor를 이용해 @TemporalWorkflow, @TemporalActivity 등이 붙은 빈을 스캔해서 Worker에 등록하는 로직을 구현할 수 있습니다.

java
코드 복사
public class TemporalBeanPostProcessor implements BeanPostProcessor {

    private final ApplicationContext context;
    private final TemporalWorkerRegistrar registrar;

    public TemporalBeanPostProcessor(ApplicationContext context, TemporalWorkerRegistrar registrar) {
        this.context = context;
        this.registrar = registrar;
    }

    @Override
    public Object postProcessAfterInitialization(Object bean, String beanName) {
        Class<?> clazz = bean.getClass();
        
        // 액티비티 구현체 등록
        if (clazz.isAnnotationPresent(TemporalActivity.class)) {
            registrar.registerActivity(bean);
        }

        // 워크플로우 구현체 등록 (WorkflowImplementationTypes)
        if (clazz.isAnnotationPresent(TemporalWorkflowImpl.class)) {
            registrar.registerWorkflowImplementationTypes(clazz);
        }

        return bean;
    }
}

DSL 지원 예제
WorkflowScenario 클래스를 이용하면 워크플로우를 메서드 체이닝 방식으로 정의할 수 있습니다. 이 DSL은 액티비티 호출, 조건문, 병렬 실행 등을 표현할 수 있도록 확장할 수 있습니다. 여기서는 간단히 액티비티 호출과 종료를 예로 듭니다.

package com.example.temporal.dsl;

import java.util.ArrayList;
import java.util.List;
import java.util.function.Consumer;
import java.util.function.Function;

import io.temporal.workflow.Workflow;

/**
 * 워크플로우 시나리오를 DSL 형태로 정의할 수 있는 클래스 예제
 */
public class WorkflowScenario<T> {

    private List<Step<T>> steps = new ArrayList<>();
    private T context;
    
    private WorkflowScenario(T initialContext) {
        this.context = initialContext;
    }
    
    public static <T> WorkflowScenario<T> start(T initialContext) {
        return new WorkflowScenario<>(initialContext);
    }
    
    public WorkflowScenario<T> activity(Consumer<T> activityCall) {
        steps.add(new ActivityStep<>(activityCall));
        return this;
    }
    
    public WorkflowScenario<T> activity(Function<T, T> activityCall) {
        steps.add(new ActivityStepWithResult<>(activityCall));
        return this;
    }
    
    public void end() {
        // 워크플로우 내에서 steps를 순차적으로 실행하는 로직
        for (Step<T> step : steps) {
            context = step.execute(context);
        }
    }
    
    interface Step<T> {
        T execute(T context);
    }
    
    static class ActivityStep<T> implements Step<T> {
        private Consumer<T> activity;
        
        public ActivityStep(Consumer<T> activity) {
            this.activity = activity;
        }

        @Override
        public T execute(T context) {
            // 여기에 Temporal의 Activity Stub 호출 로직을 연결 가능
            activity.accept(context);
            return context;
        }
    }
    
    static class ActivityStepWithResult<T> implements Step<T> {
        private Function<T, T> activity;
        
        public ActivityStepWithResult(Function<T, T> activity) {
            this.activity = activity;
        }

        @Override
        public T execute(T context) {
            return activity.apply(context);
        }
    }
}

SAGA 패턴 템플릿 예제
AbstractSagaWorkflow 클래스를 상속하여 SAGA 패턴을 간단하게 구현할 수 있습니다. 각 단계(Step)와 보상(Compensation)을 등록하고, 실행 중 오류가 나면 자동으로 보상 트랜잭션이 수행됩니다.

추상 클래스 정의
java
코드 복사
package com.example.temporal.saga;

import java.util.ArrayList;
import java.util.List;

import io.temporal.workflow.Workflow;

/**
 * SAGA 패턴을 쉽게 구현할 수 있는 추상 워크플로우 클래스 예제
 */
public abstract class AbstractSagaWorkflow {

    private final List<SagaStep> steps = new ArrayList<>();
    private int currentStepIndex = -1;

    protected void addStep(Runnable stepAction, Runnable compensationAction) {
        steps.add(new SagaStep(stepAction, compensationAction));
    }

    public void execute() {
        try {
            for (int i = 0; i < steps.size(); i++) {
                currentStepIndex = i;
                steps.get(i).stepAction.run();
            }
        } catch (Exception e) {
            compensate();
            throw e;
        }
    }

    private void compensate() {
        // 현재까지 성공한 단계에 대해 역순으로 보상 실행
        for (int i = currentStepIndex; i >= 0; i--) {
            try {
                steps.get(i).compensationAction.run();
            } catch (Exception ce) {
                Workflow.getLogger(getClass()).error("Compensation failed for step " + i, ce);
            }
        }
    }

    static class SagaStep {
        Runnable stepAction;
        Runnable compensationAction;

        public SagaStep(Runnable stepAction, Runnable compensationAction) {
            this.stepAction = stepAction;
            this.compensationAction = compensationAction;
        }
    }
}

아이디어 개요
@SagaWorkflow: 이 클래스를 SAGA 워크플로우로 정의한다는 것을 표시.
@SagaStep / @CompensationStep: 이 메서드들이 정상 처리 단계와 보상 단계임을 지정.
@WorkflowMethod: Temporal 워크플로우 진입점(메인 메서드)을 표시.
라이브러리는 Spring Bean 후처리기나 Annotation Processor 등을 활용해서 다음을 자동으로 수행합니다.

@SagaWorkflow 어노테이션이 붙은 클래스에서 @SagaStep, @CompensationStep 메서드 매핑을 추출.
내부적으로 AbstractSagaWorkflow를 상속하는 동적 프록시나 생성된 클래스를 만들어 execute() 호출 시 SAGA 로직을 수행.
@WorkflowInterface, @WorkflowMethod가 요구하는 인터페이스를 런타임 또는 컴파일타임에 자동 생성하거나, 인터페이스 없이도 워크플로우로 등록할 수 있는 방식을 지원(동적 프록시).

라이브러리 내부 구현 (개념적 예제)
이 로직은 대략적으로 다음과 같은 과정을 거칩니다.

Spring Bean Post Processor 또는 Annotation Processor를 통해 @SagaWorkflow가 붙은 클래스를 스캔.
해당 클래스의 @WorkflowMethod가 붙은 메서드를 찾아 워크플로우 시작점으로 설정.
@SagaStep, @CompensationStep 어노테이션이 붙은 메서드를 페어링해서 SAGA 단계 리스트를 구성.
내부적으로 AbstractSagaWorkflow와 유사한 로직을 가진 동적 프록시 객체를 생성하거나, 리플렉션으로 SAGA 단계를 관리하는 객체를 만듦.
Temporal에서 해당 워크플로우를 시작하면, 프록시/동적 객체가 execute()를 호출하며 단계별로 SAGA를 수행하고, 실패 시 보상 단계들을 역순으로 실행.
java
코드 복사
package com.example.temporal.internal;

import com.example.temporal.annotations.*;
import com.example.temporal.saga.AbstractSagaWorkflow;

import java.lang.reflect.Method;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class SagaWorkflowRegistry {

    private final Map<Class<?>, SagaWorkflowDefinition> sagaDefinitions = new HashMap<>();

    public void registerSagaWorkflow(Class<?> clazz) {
        if (!clazz.isAnnotationPresent(SagaWorkflow.class)) return;

        Method workflowMethod = null;
        List<Method> sagaSteps = new ArrayList<>();
        List<Method> compensationSteps = new ArrayList<>();

        for (Method m : clazz.getMethods()) {
            if (m.isAnnotationPresent(WorkflowMethod.class)) {
                workflowMethod = m;
            } else if (m.isAnnotationPresent(SagaStep.class)) {
                sagaSteps.add(m);
            } else if (m.isAnnotationPresent(CompensationStep.class)) {
                compensationSteps.add(m);
            }
        }

        // 여기서는 단순히 sagaSteps, compensationSteps를 
        // 인덱스 순서대로 매칭한다고 가정한다. (실제 구현에서는 id나 naming 규칙 필요)
        if (workflowMethod == null || sagaSteps.size() != compensationSteps.size()) {
            throw new RuntimeException("Invalid saga workflow definition");
        }

        sagaDefinitions.put(clazz, new SagaWorkflowDefinition(workflowMethod, sagaSteps, compensationSteps));
    }

    public AbstractSagaWorkflow createSagaInstance(Class<?> clazz, Object targetInstance) {
        SagaWorkflowDefinition def = sagaDefinitions.get(clazz);
        if (def == null) {
            throw new RuntimeException("No saga definition found for class " + clazz);
        }

        return new ReflectionBasedSagaWorkflow(def, targetInstance);
    }

    static class SagaWorkflowDefinition {
        Method workflowMethod;
        List<Method> sagaSteps;
        List<Method> compensationSteps;

        public SagaWorkflowDefinition(Method workflowMethod, List<Method> sagaSteps, List<Method> compensationSteps) {
            this.workflowMethod = workflowMethod;
            this.sagaSteps = sagaSteps;
            this.compensationSteps = compensationSteps;
        }
    }

    static class ReflectionBasedSagaWorkflow extends AbstractSagaWorkflow {

        private final SagaWorkflowDefinition def;
        private final Object targetInstance;

        public ReflectionBasedSagaWorkflow(SagaWorkflowDefinition def, Object targetInstance) {
            this.def = def;
            this.targetInstance = targetInstance;
        }

        @Override
        public void runSaga(String orderId) {
            // runSaga 호출 시 SAGA 단계를 등록하고 실행
            for (int i = 0; i < def.sagaSteps.size(); i++) {
                Method stepMethod = def.sagaSteps.get(i);
                Method compMethod = def.compensationSteps.get(i);

                addStep(
                    () -> invokeMethod(stepMethod, orderId),
                    () -> invokeMethod(compMethod, orderId)
                );
            }

            execute();
        }

        private void invokeMethod(Method method, String orderId) {
            try {
                method.invoke(targetInstance, orderId);
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
        }
    }
}

    String[] order(); // 실행할 SAGA 스텝의 이름 순서

@Retention(RetentionPolicy.RUNTIME)
@Target(ElementType.METHOD)

여기서 @WorkflowMethod 어노테이션은 더 이상 사용자 코드에 붙이지 않고, 라이브러리에서 동적으로 생성하는 인터페이스에 붙일 예정입니다. 즉, 라이브러리가 자동으로 워크플로우 인터페이스 및 구현을 만들어 runSaga(String orderId)에 @WorkflowMethod를 부여할 것입니다.

SagaWorkflowRegistry 수정
이전 예제에서 인덱스 매칭 대신 이름 매칭을 하고, runSaga 메서드를 동적으로 가진 워크플로우 인터페이스/구현을 생성합니다.

java
코드 복사
package com.example.temporal.internal;

import com.example.temporal.annotations.*;
import io.temporal.workflow.WorkflowMethod;
import io.temporal.workflow.WorkflowInterface;
import com.example.temporal.saga.AbstractSagaWorkflow;

import java.lang.reflect.Method;
import java.util.*;
import java.util.stream.Collectors;

/**
 * SAGA 워크플로우 정의를 스캔하고, 이름 기반으로 Steps를 매칭하며,
 * order에 따라 SAGA를 실행하는 동적 구현을 생성하는 레지스트리 예제
 */
public class SagaWorkflowRegistry {

    private final Map<Class<?>, SagaWorkflowDefinition> sagaDefinitions = new HashMap<>();

    public void registerSagaWorkflow(Class<?> clazz) {
        if (!clazz.isAnnotationPresent(SagaWorkflow.class)) return;

        SagaWorkflow sagaAnno = clazz.getAnnotation(SagaWorkflow.class);
        String[] order = sagaAnno.order();
        Set<String> requiredNames = new LinkedHashSet<>(Arrays.asList(order));

        Map<String, Method> stepsMap = new HashMap<>();
        Map<String, Method> compensationMap = new HashMap<>();

        for (Method m : clazz.getMethods()) {
            if (m.isAnnotationPresent(SagaStep.class)) {
                String name = m.getAnnotation(SagaStep.class).name();
                stepsMap.put(name, m);
            } else if (m.isAnnotationPresent(CompensationStep.class)) {
                String name = m.getAnnotation(CompensationStep.class).name();
                compensationMap.put(name, m);
            }
        }

        // order에 명시된 모든 step name에 대해 step과 compensation이 모두 존재해야 함
        for (String stepName : requiredNames) {
            if (!stepsMap.containsKey(stepName) || !compensationMap.containsKey(stepName)) {
                throw new RuntimeException("Missing step or compensation for name: " + stepName);
            }
        }

        sagaDefinitions.put(clazz, new SagaWorkflowDefinition(order, stepsMap, compensationMap));
    }

    public AbstractSagaWorkflow createSagaInstance(Class<?> clazz, Object targetInstance) {
        SagaWorkflowDefinition def = sagaDefinitions.get(clazz);
        if (def == null) {
            throw new RuntimeException("No saga definition found for class " + clazz);
        }

        return new ReflectionBasedSagaWorkflow(def, targetInstance);
    }

    static class SagaWorkflowDefinition {
        String[] order;
        Map<String, Method> stepsMap;
        Map<String, Method> compensationMap;

        public SagaWorkflowDefinition(String[] order, Map<String, Method> stepsMap, Map<String, Method> compensationMap) {
            this.order = order;
            this.stepsMap = stepsMap;
            this.compensationMap = compensationMap;
        }
    }

    static class ReflectionBasedSagaWorkflow extends AbstractSagaWorkflow {

        private final SagaWorkflowDefinition def;
        private final Object targetInstance;

        public ReflectionBasedSagaWorkflow(SagaWorkflowDefinition def, Object targetInstance) {
            this.def = def;
            this.targetInstance = targetInstance;
        }

        // 동적 runSaga 메서드를 여기서 구현:
        // 실제로 Temporal에선 인터페이스에 @WorkflowMethod 붙은 메서드가 필요하므로,
        // 라이브러리 초기화 시점에 프록시 인터페이스/구현을 만들어 여기에 runSaga를 제공하거나,
        // 혹은 공용 인터페이스 (e.g. SagaRunner) 를 사용한 뒤 이 메서드에서 실행.
        @Override
        public void runSaga(String orderId) {
            // order 순서대로 step, compensation 등록
            for (String stepName : def.order) {
                Method stepMethod = def.stepsMap.get(stepName);
                Method compMethod = def.compensationMap.get(stepName);

                addStep(
                    () -> invokeMethod(stepMethod, orderId),
                    () -> invokeMethod(compMethod, orderId)
                );
            }

            // 모든 스텝 실행
            execute();
        }

        private void invokeMethod(Method method, String orderId) {
            try {
                method.invoke(targetInstance, orderId);
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
        }
    }
}
동적 인터페이스/구현 생성
Temporal 워크플로우를 시작하기 위해서는 @WorkflowInterface와 @WorkflowMethod가 달린 인터페이스가 필요합니다. 런타임에 이를 동적으로 생성하는 방식 중 하나는 다음과 같습니다.

하나의 공용 인터페이스 예: SagaRunner 인터페이스를 미리 정의해두고, 여기엔 @WorkflowInterface, @WorkflowMethod가 붙은 runSaga(String orderId) 메서드가 있음.
라이브러리가 SagaWorkflowRegistry에서 생성한 AbstractSagaWorkflow 인스턴스를 내부적으로 가지고 있는 프록시 구현체를 만들고, runSaga 호출 시 해당 인스턴스를 호출.
예를 들어:

package com.example.temporal.internal;

import java.lang.reflect.Proxy;

public class SagaRunnerFactory {

    private final SagaWorkflowRegistry registry;

    public SagaRunnerFactory(SagaWorkflowRegistry registry) {
        this.registry = registry;
    }

    public SagaRunner createRunner(Class<?> sagaDefinitionClass, Object sagaDefinitionInstance) {
        AbstractSagaWorkflow saga = registry.createSagaInstance(sagaDefinitionClass, sagaDefinitionInstance);
        return (SagaRunner) Proxy.newProxyInstance(
            SagaRunner.class.getClassLoader(),
            new Class<?>[]{SagaRunner.class},
            (proxy, method, args) -> {
                if ("runSaga".equals(method.getName())) {
                    saga.runSaga((String) args[0]);
                    return null;
                }
                return null;
            }
        );
    }
}

변경 사항 개요
자유로운 파라미터 지원:
각 @SagaStep 및 @CompensationStep 메서드는 (String orderId), (String orderId, int amount) 등 임의의 파라미터를 가질 수 있습니다.

파라미터 바인딩:
@SagaParam("키이름") 어노테이션을 추가하여, 각 파라미터에 매칭될 값을 runSaga(Map<String,Object> params) 호출 시 전달된 파라미터 맵에서 가져오도록 합니다.

runSaga 파라미터 변경:
runSaga(Map<String,Object> params) 형태로 변경하여, SAGA 실행 시 필요한 모든 인자를 key-value 맵 형태로 전달할 수 있게 합니다.

이렇게 하면 개발자는 SAGA Step 메서드 파라미터에 원하는 타입과 개수를 선언하고, @SagaParam("...")로 컨텍스트에서 값을 주입받을 수 있습니다.

SagaWorkflowRegistry
이전과 유사하지만, 파라미터를 주입하기 위해 @SagaParam를 분석하고, 실제 호출 시 파라미터 맵에서 값을 꺼내어 메서드에 주입하는 로직을 추가합니다.

java
코드 복사
package com.example.temporal.internal;

import com.example.temporal.annotations.*;
import com.example.temporal.saga.AbstractSagaWorkflow;

import java.lang.reflect.Method;
import java.lang.reflect.Parameter;
import java.util.*;

public class SagaWorkflowRegistry {

    private final Map<Class<?>, SagaWorkflowDefinition> sagaDefinitions = new HashMap<>();

    public void registerSagaWorkflow(Class<?> clazz) {
        if (!clazz.isAnnotationPresent(SagaWorkflow.class)) return;

        SagaWorkflow sagaAnno = clazz.getAnnotation(SagaWorkflow.class);
        String[] order = sagaAnno.order();
        Set<String> requiredNames = new LinkedHashSet<>(Arrays.asList(order));

        Map<String, StepDefinition> stepsMap = new HashMap<>();
        Map<String, StepDefinition> compensationMap = new HashMap<>();

        for (Method m : clazz.getMethods()) {
            SagaStep stepAnno = m.getAnnotation(SagaStep.class);
            CompensationStep compAnno = m.getAnnotation(CompensationStep.class);

            if (stepAnno != null) {
                stepsMap.put(stepAnno.name(), new StepDefinition(stepAnno.name(), m));
            }
            if (compAnno != null) {
                compensationMap.put(compAnno.name(), new StepDefinition(compAnno.name(), m));
            }
        }

        // 검증: order에 명시된 모든 step name에 대해 step과 compensation이 존재해야 함
        for (String stepName : requiredNames) {
            if (!stepsMap.containsKey(stepName) || !compensationMap.containsKey(stepName)) {
                throw new RuntimeException("Missing step or compensation for name: " + stepName);
            }
        }

        sagaDefinitions.put(clazz, new SagaWorkflowDefinition(order, stepsMap, compensationMap));
    }

    public AbstractSagaWorkflow createSagaInstance(Class<?> clazz, Object targetInstance) {
        SagaWorkflowDefinition def = sagaDefinitions.get(clazz);
        if (def == null) {
            throw new RuntimeException("No saga definition found for class " + clazz);
        }

        return new ReflectionBasedSagaWorkflow(def, targetInstance);
    }

    static class SagaWorkflowDefinition {
        String[] order;
        Map<String, StepDefinition> stepsMap;
        Map<String, StepDefinition> compensationMap;

        public SagaWorkflowDefinition(String[] order, Map<String, StepDefinition> stepsMap, Map<String, StepDefinition> compensationMap) {
            this.order = order;
            this.stepsMap = stepsMap;
            this.compensationMap = compensationMap;
        }
    }

    static class StepDefinition {
        String name;
        Method method;
        ParameterDefinition[] parameters;

        public StepDefinition(String name, Method method) {
            this.name = name;
            this.method = method;
            this.parameters = parseParameters(method);
        }

        private ParameterDefinition[] parseParameters(Method m) {
            Parameter[] params = m.getParameters();
            ParameterDefinition[] pdefs = new ParameterDefinition[params.length];
            for (int i = 0; i < params.length; i++) {
                Parameter p = params[i];
                SagaParam paramAnno = p.getAnnotation(SagaParam.class);
                if (paramAnno == null) {
                    throw new RuntimeException("All parameters must be annotated with @SagaParam");
                }
                pdefs[i] = new ParameterDefinition(paramAnno.value(), p.getType());
            }
            return pdefs;
        }

        public Object[] resolveParameters(Map<String, Object> context) {
            Object[] args = new Object[parameters.length];
            for (int i = 0; i < parameters.length; i++) {
                ParameterDefinition pdef = parameters[i];
                Object value = context.get(pdef.key);
                if (value == null) {
                    throw new RuntimeException("Missing parameter: " + pdef.key);
                }
                // 타입 변환 로직이 필요할 수 있으나 여기서는 단순 캐스팅 가정
                if (!pdef.paramType.isInstance(value)) {
                    throw new RuntimeException("Parameter " + pdef.key + " is not of type " + pdef.paramType);
                }
                args[i] = value;
            }
            return args;
        }
    }

    static class ParameterDefinition {
        String key;
        Class<?> paramType;

        public ParameterDefinition(String key, Class<?> paramType) {
            this.key = key;
            this.paramType = paramType;
        }
    }

    static class ReflectionBasedSagaWorkflow extends AbstractSagaWorkflow {

        private final SagaWorkflowDefinition def;
        private final Object targetInstance;

        public ReflectionBasedSagaWorkflow(SagaWorkflowDefinition def, Object targetInstance) {
            this.def = def;
            this.targetInstance = targetInstance;
        }

        // runSaga를 대신할 메서드: 이제 runSaga(Map<String,Object> params) 형태로 가정
        public void runSaga(Map<String,Object> params) {
            for (String stepName : def.order) {
                StepDefinition stepDef = def.stepsMap.get(stepName);
                StepDefinition compDef = def.compensationMap.get(stepName);

                addStep(
                    () -> invokeMethod(stepDef, params),
                    () -> invokeMethod(compDef, params)
                );
            }

            execute();
        }

        private void invokeMethod(StepDefinition stepDef, Map<String,Object> params) {
            try {
                Object[] args = stepDef.resolveParameters(params);
                stepDef.method.invoke(targetInstance, args);
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
        }
    }
}

package com.example.temporal.internal;

import java.lang.reflect.Proxy;
import java.util.Map;

public class SagaRunnerFactory {
    private final SagaWorkflowRegistry registry;

    public SagaRunnerFactory(SagaWorkflowRegistry registry) {
        this.registry = registry;
    }

    public SagaRunner createRunner(Class<?> sagaDefinitionClass, Object sagaDefinitionInstance) {
        AbstractSagaWorkflow saga = registry.createSagaInstance(sagaDefinitionClass, sagaDefinitionInstance);
        return (SagaRunner) Proxy.newProxyInstance(
            SagaRunner.class.getClassLoader(),
            new Class<?>[]{SagaRunner.class},
            (proxy, method, args) -> {
                if ("runSaga".equals(method.getName())) {
                    // args[0]은 Map<String,Object>로 가정
                    Map<String,Object> params = (Map<String,Object>) args[0];
                    // ReflectionBasedSagaWorkflow의 runSaga(Map) 호출
                    saga.getClass().getMethod("runSaga", Map.class).invoke(saga, params);
                    return null;
                }
                return null;
            }
        );
    }

사용자가 워크플로우 실행 시 다음과 같이 다양한 파라미터를 전달할 수 있습니다.

java
코드 복사
Map<String,Object> params = new HashMap<>();
params.put("orderId", "order-1234");
params.put("amount", 1000);
params.put("address", "123 Main St");

SagaWorkflowRegistry registry = new SagaWorkflowRegistry();
registry.registerSagaWorkflow(OrderSagaWorkflowDefinition.class);

OrderSagaWorkflowDefinition definitionInstance = new OrderSagaWorkflowDefinition();
SagaRunnerFactory runnerFactory = new SagaRunnerFactory(registry);
SagaRunner runner = runnerFactory.createRunner(OrderSagaWorkflowDefinition.class, definitionInstance);

runner.runSaga(params);

// 이 호출 시,
// 1. "reserveInventory" 스텝: orderId 파라미터 -> "order-1234"
// 2. "processPayment" 스텝: orderId -> "order-1234", amount -> 1000
// 3. "requestShipping" 스텝: orderId -> "order-1234", address -> "123 Main St"
정리
이 예제에서는:

@SagaParam 어노테이션을 통해 메서드 파라미터를 파라미터 맵으로부터 동적으로 주입.
runSaga(Map<String,Object> params) 형태로 SAGA를 실행할 때 여러 파라미터를 전달 가능.
@SagaStep, @CompensationStep에 name을 지정하고, @SagaWorkflow에 order를 지정해 이름 기준으로 매칭.
결과적으로 개발자는 자유롭게 파라미터를 정의할 수 있고, 라이브러리는 이를 바탕으로 동적 바인딩을 수행하여 SAGA를 실행.
이로써 상속 및 구현 없이, 어노테이션과 파라미터 매핑만으로 복잡하고 유연한 SAGA 워크플로우를 정의하는 방법을 제시할 수 있습니다.
